' ============================================================
' Programa 5.1
' Estimacion recursiva del test de Chow
' (pp. 348-349)
' ============================================================
' Programa para detectar la presencia de quiebre
' estructural sobre la base del test de Chow
' ============================================================
'USO:
'El nombre de la variable dependiente
'debera pasarse como el primer argumento
'del programa.
'Debera crearse un grupo Indep que contenga
'todos los regresores del modelo.
' ============================================================

TABLE(3,4) Resultados
Resultados(1,1) = "Periodo"
Resultados(1,2) = "Fecha"
Resultados(1,3) = "F_Stat"
Resultados(1,4) = "F_Prob"
SETLINE(Resultados,2)

SMPL @ALL

GENR _Dep = %0
!Obs = @OBS(_Dep)
%Inicio = @OTOD(1)
%Fin = @OTOD(!Obs)

EQUATION Restringida.LS _Dep c Indep
!k = Restringida.@NCOEF
!Scrr = Restringida.@SSR

!MaxF = 0
EQUATION Sinres1
EQUATION Sinres2

'Inicio del algoritmo de busqueda.
' ============================================================

FOR !n = !k+2 to @DTOO(%Fin)-(!k+2)

   %Fin2 = @OTOD(!n)
   %Inicio2 = @OTOD(!n+1)

   SMPL %Inicio %Fin2
   Sinres1.LS _Dep c Indep
   !Scrs1 = Sinres1.@SSR

   SMPL %Inicio2 %Fin
   Sinres2.LS _Dep c Indep
   !Scrs2 = Sinres2.@SSR

   !F = ((!Scrr-(!Scrs1+!Scrs2))/(!k)) / ((!Scrs1+!Scrs2)/(!Obs-2*!k))

   IF !F > !MaxF THEN
      !MaxF = !F
      !Periodo = !n
      %Fecha = @OTOD(!n)
   ENDIF

NEXT

SETCELL(Resultados,3,1,!Periodo,0)
Resultados(3,2) = %Fecha
Resultados(3,3) = !MaxF
Resultados(3,4) = @FDIST(!MaxF,!k,!Obs-2*!k)

SMPL @ALL

D _Dep
D Sinres1
D Sinres2
D Restringida

'Fin del programa.
